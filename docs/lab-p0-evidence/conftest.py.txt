@dataclass
class Clock:
    """Injizierbare Uhr: Tests setzen `now` explizit, nie die Systemzeit."""

    now: datetime

    def set(self, iso: str) -> "Clock":
        """Setzt die Uhr auf einen ISO-8601-Zeitpunkt (mit Offset oder 'Z')."""
        self.now = datetime.fromisoformat(iso.replace("Z", "+00:00"))
        if self.now.tzinfo is None:
            raise ValueError("Clock.set erwartet einen zeitzonenbewussten Zeitpunkt")
        return self

    def advance(self, **kwargs) -> "Clock":
        self.now = self.now + timedelta(**kwargs)
        return self

    @property
    def utc(self) -> datetime:
        return self.now.astimezone(timezone.utc)

    @property
    def local(self) -> datetime:
        return self.now.astimezone(ZONE)
