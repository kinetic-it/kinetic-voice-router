"""P0-Selbsttest der Testbasis: 10 Slots, injizierte Uhr, DST — ohne HA.

Diese Tests pruefen nur das Geruest (P0). Sie aendern weder Storelogik noch
Produktivcode und laufen ohne Home-Assistant-Installation.
"""
from __future__ import annotations

import sys
from datetime import timedelta, timezone

from zoneinfo import ZoneInfo

ZONE = ZoneInfo("Europe/Berlin")
CET = timedelta(hours=1)
CEST = timedelta(hours=2)


# --- 10 Slots -------------------------------------------------------------


def test_zehn_slots_definiert(slots):
    assert slots == tuple(range(1, 11))
    assert len(slots) == 10
    assert len(set(slots)) == 10


def test_produktive_slots_sind_teilmenge(slots):
    # Ist-Stand: Slots 1..3 produktiv; P1..P5 muessen auf 1..10 erweitern.
    assert {1, 2, 3}.issubset(set(slots))


# --- Kein Home Assistant --------------------------------------------------


def test_kern_importiert_ohne_home_assistant(core):
    assert hasattr(core, "Store")
    assert "homeassistant" not in sys.modules


# --- Injizierte Uhr -------------------------------------------------------


def test_uhr_ist_injizierbar_und_nicht_die_systemzeit(clock):
    assert clock.local.hour == 8
    assert clock.local.date().isoformat() == "2026-11-18"
    clock.set("2026-03-29T00:00:00+01:00")
    assert clock.utc.date().isoformat() == "2026-03-28"
    assert clock.utc.hour == 23


def test_uhr_vorlauf_ist_deterministisch(clock):
    start = clock.utc
    clock.advance(minutes=90)
    assert clock.utc - start == timedelta(minutes=90)


def test_store_nutzt_nur_temporaere_db(store, core):
    assert isinstance(store, core.Store)


# --- DST-Fixtures ---------------------------------------------------------


def test_dst_offsets_jahreszeiten(dst_spring, dst_fall):
    # 29.03.2026 00:00 noch CET, 25.10.2026 00:00 noch CEST.
    assert dst_spring.utcoffset() == CET
    assert dst_fall.utcoffset() == CEST
    assert dst_spring.replace(month=7, day=1, hour=12).utcoffset() == CEST
    assert dst_fall.replace(month=1, day=15, hour=12).utcoffset() == CET


def test_dst_fruehlingsluecke_ist_nicht_existent(dst_local_times):
    naive = dst_local_times["spring_gap"]
    possible = set()
    for fold in (0, 1):
        utc = naive.replace(tzinfo=ZONE, fold=fold).astimezone(timezone.utc)
        if utc.astimezone(ZONE).replace(tzinfo=None) == naive:
            possible.add(utc)
    assert possible == set()


def test_dst_herbststunde_ist_mehrdeutig(dst_local_times):
    naive = dst_local_times["fall_ambiguous"]
    possible = set()
    for fold in (0, 1):
        utc = naive.replace(tzinfo=ZONE, fold=fold).astimezone(timezone.utc)
        if utc.astimezone(ZONE).replace(tzinfo=None) == naive:
            possible.add(utc)
    assert len(possible) == 2


def test_kern_legacy_hilfe_ueberspringt_dst_luecke(core, clock):
    # 02:30 existiert am 29.03.2026 nicht -> vorheriger Tag (28.03.) gewinnt.
    clock.set("2026-03-29T06:00:00+02:00")
    occ = core.legacy_last_occurrence(2, "02:30", clock.utc)
    assert occ is not None
    assert occ <= clock.utc
    local = occ.astimezone(ZONE)
    assert local.strftime("%H:%M") == "02:30"
    assert local.date().isoformat() == "2026-03-28"
